{{- /* Generic shortcode for per-post bespoke widgets. Looks up .js and .css from the post's page bundle resources and emits them once. Any extra named params (other than `name`) become data-* attributes on the mount div, so widget JS can read config. Usage in a post bundle: content/posts/foo/ ├── index.md ├── widget.js ← self-init: querySelectorAll('[data-embed="widget"]') └── widget.css ← optional Markdown: {{< embed name="widget" color="red" >}} */ -}} {{- $name := .Get "name" -}} {{- if not $name -}}{{- errorf "embed: missing required 'name' param at %s" .Position -}}{{- end -}} {{- $jsRes := .Page.Resources.GetMatch (printf "%s.js" $name) -}} {{- $cssRes := .Page.Resources.GetMatch (printf "%s.css" $name) -}} {{- $loadedKey := printf "embed_%s_loaded" $name -}}
{{- if not (.Page.Scratch.Get $loadedKey) -}} {{- with $cssRes -}} {{- $css := . | minify | fingerprint -}} {{- end -}} {{- with $jsRes -}} {{- $js := . | minify | fingerprint -}} {{- end -}} {{- .Page.Scratch.Set $loadedKey true -}} {{- end -}}
